
@ratelimit(key='ip', rate='5/m', block=True)
def login_view(request):
    if request.user.is_authenticated:
        return redirect('home')

    if request.method == 'POST':
        username = request.POST.get('username', ' ').strip()
        password = request.POST.get('password', ' ').strip()

        user = authenticate(request, username=username, password=password)
        print ("Intentando autenticar:",user.rol )
        if user is not None:
            login(request, user)

            # 🔑 DOCENTE → centro fijo
            if user.rol == 'docente' and hasattr(user, 'docente'):
                request.session['centro_id'] = user.docente.centro_id
            if user.rol == 'director' and hasattr(user, 'administrativo'):
                        request.session['centro_id'] = user.administrativo.centro_id
            if user.rol == 'estudiante' and hasattr(user, 'estudiante'):
                request.session['centro_id'] = user.estudiante.centro_id
            if user.rol == 'administrativo' and hasattr(user, 'administrativo'):
                request.session['centro_id'] = user.administrativo.centro_id
            if user.rol == 'superadmin' and hasattr(user, 'admin'):
                request.session['centro_id'] = user.admin.centro_id
            

            return redirect('home')

        return render(request, 'usuarios/login.html', {
            'error': 'Usuario o contraseña incorrectos'
        })

    return render(request, 'usuarios/login.html')
